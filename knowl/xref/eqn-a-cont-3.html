\begin{align} \left| \int_{\Gamma_-} (\mathbf{b} \cdot \mathbf{n}) \, w v \, \mathrm{d}S \right| &\leq \int_{\Gamma_-} |\mathbf{b} \cdot \mathbf{n}| \, |w| |v| \, \mathrm{d}S\tag{1.5.44}\\ &\leq \left( \int_{\Gamma_-} |\mathbf{b}\cdot\mathbf{n}|\,w^2\,\mathrm{d}S \right)^{1/2} \left( \int_{\Gamma_-} |\mathbf{b}\cdot\mathbf{n}|\,v^2\,\mathrm{d}S \right)^{1/2}\tag{1.5.45}\\ &\leq \left( \int_{\partial\Omega} |\mathbf{b}\cdot\mathbf{n}|\,w^2\,\mathrm{d}S \right)^{1/2} \left( \int_{\partial\Omega} |\mathbf{b}\cdot\mathbf{n}|\,v^2\,\mathrm{d}S \right)^{1/2}\tag{1.5.46}\\ &\leq \enorm{w} \enorm{v}\tag{1.5.47}\\ &\leq \snorm{w} \enorm{v}\tag{1.5.48} \end{align}
